cmake_minimum_required(VERSION 3.24)
project(filtered_baselines LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(FAISS_ENABLE_GPU OFF CACHE BOOL "" FORCE)
set(FAISS_ENABLE_PYTHON OFF CACHE BOOL "" FORCE)
set(FAISS_ENABLE_MKL OFF CACHE BOOL "" FORCE)
set(FAISS_ENABLE_EXTRAS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(FAISS_OPT_LEVEL generic CACHE STRING "" FORCE)
set(BLA_VENDOR OpenBLAS CACHE STRING "")
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG -march=native")
if(NOT FANN_ALGORITHM MATCHES "^(navix|acorn)$")
    message(FATAL_ERROR "FANN_ALGORITHM must be navix or acorn")
endif()
add_subdirectory("${FANN_SOURCE}" upstream EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)
target_compile_definitions(faiss PRIVATE NDEBUG)
add_executable(filtered_benchmark benchmark.cc)
target_include_directories(filtered_benchmark PRIVATE "${FANN_SOURCE}")
target_compile_definitions(filtered_benchmark PRIVATE "FANN_${FANN_ALGORITHM}=1")
target_compile_options(filtered_benchmark PRIVATE -Wall -Wextra)
target_link_libraries(filtered_benchmark PRIVATE faiss Threads::Threads OpenMP::OpenMP_CXX)
